def band(v,low,high):
 h=(1-np.exp(-(f/low)**4))*np.exp(-(f/high)**4);return np.fft.irfft(np.fft.rfft(v)*h,n=len(v))
def rms(v):return np.sqrt(np.mean(v*v))
def unit(v):return v/max(rms(v),1e-9)
def stats(v):

 return {'rmsDb':float(20*np.log10(rms(v))),'peak':float(abs(v).max()),'above200Percent':float(p[f>=200].sum()/p.sum()*100),'seamDelta':float(abs(v[0]-v[-1]))}
