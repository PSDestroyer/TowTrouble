if(UnityEditor.EditorApplication.isPlayingOrWillChangePlaymode)throw new System.Exception("Edit Mode required");
var rows=new System.Collections.Generic.List<object>();
foreach(string name in new[]{"Engine Idle","Engine Drive"}){
var path="Assets/TowTrouble/Art/Audio/Library/Vehicles/"+name+".wav";UnityEditor.AssetDatabase.ImportAsset(path,UnityEditor.ImportAssetOptions.ForceUpdate|UnityEditor.ImportAssetOptions.ForceSynchronousImport);
var clip=UnityEditor.AssetDatabase.LoadAssetAtPath<AudioClip>(path);clip.LoadAudioData();var samples=new float[clip.samples*clip.channels];if(!clip.GetData(samples,0))throw new System.Exception("No imported PCM");
float peak=samples.Max(v=>Mathf.Abs(v));double sum=samples.Sum(v=>(double)v*v);if(peak>.95f||sum/samples.Length<.005)throw new System.Exception("Unexpected imported engine signal");
var importer=(UnityEditor.AudioImporter)UnityEditor.AssetImporter.GetAtPath(path);
rows.Add(new{name,clip.frequency,clip.channels,clip.length,peak,rmsDb=10*System.Math.Log10(sum/samples.Length),loadType=importer.defaultSampleSettings.loadType.ToString(),format=importer.defaultSampleSettings.compressionFormat.ToString()});
}
var bank=UnityEditor.AssetDatabase.LoadAssetAtPath<TowTrouble.RecoveryAudioBank>("Assets/TowTrouble/Art/Audio/Resources/RecoveryAudioBank.asset");
if(bank.EngineLoad!=bank.EngineDrive||bank.EngineLoad==bank.WinchMotor)throw new System.Exception("Engine bank references wrong");
return new{rows,bank.EngineBoostDb,bank.EngineGain,sceneDirty=UnityEngine.SceneManagement.SceneManager.GetActiveScene().isDirty};
