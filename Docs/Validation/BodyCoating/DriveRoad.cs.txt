    UnityEditor.EditorApplication.update-=loop;input.SetValue(s,Vector2.zero);brake.SetValue(s,true);Time.timeScale=1;
    System.IO.File.WriteAllText("/private/tmp/tow-body-coating/RoadDrive.json",Newtonsoft.Json.JsonConvert.SerializeObject(new{zone,complete=error==null,error,seconds=Time.time-start,distance,rows}));

{
    try
    {
        if(!Application.isPlaying||Time.frameCount==lastFrame)return;lastFrame=Time.frameCount;
        if((System.DateTime.UtcNow-wall).TotalSeconds>180){Stop("Wall timeout");return;}

        {

            var hits=Enumerable.Range(0,4).Select(i=>{bool yes=car.TryGetWheelContact(i,out var h);return new{i,yes,collider=h.collider?.name,surface=TowTrouble.RecoveryBodyCoatingRules.Classify(h,false).ToString()};}).ToArray();
